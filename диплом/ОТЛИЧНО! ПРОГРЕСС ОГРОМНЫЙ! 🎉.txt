## 🔴 **ПРОБЛЕМА 2: UI - ссылка не кликается**

**Ошибка:** `URL did not change after clicking meal plan link`

**Причина:** Ссылка на планы питания **уже активна** (вы уже на `/mealplan`), и повторный клик не меняет URL.

**Решение - исправьте `header_component.py`:**

```python
def go_to_meal_plan(self):
    """Перейти на страницу планов питания"""
    try:
        # Проверяем, не находимся ли мы уже на странице планов
        current_url = self.driver.current_url
        if '/mealplan' in current_url:
            print(f"Already on meal plan page: {current_url}")
            return True
            
        # Ищем ссылку на планы питания
        print(f"Current URL: {current_url}")
        
        for attempt in range(5):
            print(f"Attempt {attempt + 1} to find meal plan link")
            
            try:
                # Ждём появления хедера
                header = WebDriverWait(self.driver, 10).until(
                    EC.presence_of_element_located((By.TAG_NAME, "header"))
                )
                print("Found header")
                
                # Ищем ссылку внутри хедера
                meal_plan_link = header.find_element(By.CSS_SELECTOR, "a[href*='mealplan']")
                print(f"Found meal plan link: {meal_plan_link.text}")
                print(f"Link href: {meal_plan_link.get_attribute('href')}")
                print(f"Link classes: {meal_plan_link.get_attribute('class')}")
                
                # Пробуем кликнуть
                self.driver.execute_script("arguments[0].scrollIntoView(true);", meal_plan_link)
                time.sleep(0.5)
                self.driver.execute_script("arguments[0].click();", meal_plan_link)
                
                # Ждём изменения URL
                WebDriverWait(self.driver, 10).until(
                    lambda d: '/mealplan' in d.current_url and d.current_url != current_url
                )
                print("URL changed successfully")
                return True
                
            except Exception as e:
                print(f"Attempt {attempt + 1} failed: {e}")
                if attempt < 4:
                    time.sleep(2)
                continue
        
        # Если не получилось, пробуем прямой переход
        print("Trying direct navigation")
        base_url = os.getenv('BASE_URL', 'https://app.tandoor.dev')
        self.driver.get(f"{base_url}/mealplan/")
        return True
        
    except Exception as e:
        print(f"Error in go_to_meal_plan: {e}")
        raise
```
